# Range Chart Analyzer runs on the Python standard library alone.
# The three optional deps below improve the experience but are not required:
#
# PYTHON FLOOR — 2026-11-01. Every marker below reads `python_version >= "3.10"`
# because that is what this project actually supports (README: "Python ≥ 3.10,
# CI verifies 3.10/3.12"). The markers used to say 3.8 and 3.9, left over from
# before that floor was raised, and they stayed harmless only while the pinned
# minimums were low enough to install there.
#
# They are not harmless any more. Dependabot raised the minimums to current
# releases, and two of them now refuse to install below 3.10 on their own:
# Pillow 12.3 requires >=3.10 and PySide6 6.12 requires >=3.10. Left at "3.8",
# `pip install -r requirements.txt` on a 3.8 interpreter would resolve to
# "no matching distribution" instead of skipping the optional package. Keep
# each marker at or above the pinned package's own requires_python.
#
# Pillow: enables the GUI image thumbnail preview and client-side downscaling
#          of very large images.
# sv_ttk: swaps the Tkinter "clam" theme for a Windows 11 Fluent-style
#          flat/modern skin (buttons, inputs, combobox, notebook tabs).
#          Without it the app falls back to ttk's default clam theme.
# pywebview: enables `python main.py --ui modern`, a native-window wrapper
#          around the existing web frontend. On Linux the GTK backend asks for
#          WebKit2 4.1 first and only then falls back to 4.0
#          (webview/platforms/gtk.py:34-38), so install 4.1 where it exists:
#              pip install "pywebview[gtk]"
#              sudo apt install python3-gi python3-gi-cairo \
#                  gir1.2-gtk-3.0 gir1.2-webkit2-4.1
#          Without pywebview the modern mode falls back to opening the local
#          URL in the default browser.
Pillow>=12.3.0 ; python_version >= "3.10"
sv_ttk>=2.6.1  ; python_version >= "3.10"
pywebview>=6.2.1 ; python_version >= "3.10"
# PySide6 + qfluentwidgets: enable the modern Fluent desktop GUI
#   (python main.py --ui fluent). Large (~150MB); fully optional — the
#   stdlib Tkinter GUI (python main.py) remains the default and the
#   Fluent launcher falls back to it automatically if these are missing.
PySide6>=6.12.0 ; python_version >= "3.10"
PySide6-Fluent-Widgets>=1.11.3 ; python_version >= "3.10"
# PySide6-Addons — pulls in QtWebEngineWidgets for the History detail
# dialog (gui_fluent_history_detail.HistoryDetailDialog). Heavy
# (~+50MB on top of PySide6) but optional: the rest of the Fluent GUI
# works without it; double-clicking a history row will show a no-engine
# fallback message instead of the rich result render.
PySide6-Addons>=6.12.0 ; python_version >= "3.10"
# XLSX export — enables the "Export Excel" button in the Fluent GUI and
#   the `rca_core.exporter.to_xlsx` function. Optional: the rest of the
#   app (CSV / JSON / TSV) works fine without it.
openpyxl>=3.1.5 ; python_version >= "3.10"
# Token-usage charts — rendered on the Usage page. Falls back to plain
#   text tables if missing.
matplotlib>=3.10.9 ; python_version >= "3.10"
# At-rest encryption for stored API keys (rca_core/secrets_store.py):
#   cryptography provides the Fernet envelope; keyring moves the key into
#   the OS credential store. Without them the store falls back to
#   fingerprint-derived obfuscation (any local user can recompute the
#   key). Install both for real at-rest protection.
cryptography>=50.0.2 ; python_version >= "3.10"
keyring>=25.7.0 ; python_version >= "3.10"
